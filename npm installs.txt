1. Install Angular (it's probably already installed)
npm install -g @angular/cli 

2. Create new Angular project
ng new projectname

3. Run command for creating package.json
npm init

4. Install the following for the api - preferably separate folder
npm install express body-parser mongoose npm-run-all busboy --save

5. Install nodemon (server for backend)
npm install --save-dev nodemon

6. Modify package.json
"scripts": {
  "client": "ng serve --proxy-config server.conf.json",
  "server": "nodemon server.js",
  "start": "npm-run-all -p client server"
}

7. Create file server.conf.json
{
  "/api": {
    "target": "http://localhost:3000",
    "secure": false
  }
}

8. Create server.js
var express = require('express'),
  app = express(),
  port = process.env.PORT || 3000;

app.listen(port);

console.log('todo list RESTful API server started on: ' + port);
